WITH ranked_salaries AS (
    SELECT
        department,
        salary,
        DENSE_RANK() OVER (
            PARTITION BY department
            ORDER BY salary DESC
        ) AS rank
    FROM employees
)
SELECT
    department,
    salary AS largest_salary
FROM ranked_salaries
WHERE rank = 2;
